cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET esp32c6 CACHE STRING "Target chip")
set(CASTER_WHEEL "" CACHE STRING "front_left or front_right")
set_property(CACHE CASTER_WHEEL PROPERTY STRINGS front_left front_right)
if(CASTER_WHEEL STREQUAL "front_left")
    set(CASTER_WHEEL_INDEX 1)
elseif(CASTER_WHEEL STREQUAL "front_right")
    set(CASTER_WHEEL_INDEX 2)
else()
    message(FATAL_ERROR "CASTER_WHEEL must be front_left or front_right")
endif()
set(CASTER_ROLE H1 CACHE STRING "H1, H2, CENTER, SWIVEL, WHEEL, or legacy ORDINARY")
set_property(CACHE CASTER_ROLE PROPERTY STRINGS H1 H2 CENTER ORDINARY SWIVEL WHEEL)
if(CASTER_ROLE STREQUAL "H1")
    set(CASTER_NODE_ID 1)
elseif(CASTER_ROLE STREQUAL "H2")
    set(CASTER_NODE_ID 2)
elseif(CASTER_ROLE STREQUAL "CENTER")
    set(CASTER_NODE_ID 3)
elseif(CASTER_ROLE STREQUAL "ORDINARY")
    set(CASTER_NODE_ID 4)
elseif(CASTER_ROLE STREQUAL "SWIVEL")
    set(CASTER_NODE_ID 5)
elseif(CASTER_ROLE STREQUAL "WHEEL")
    set(CASTER_NODE_ID 6)
else()
    message(FATAL_ERROR "CASTER_ROLE must be H1, H2, CENTER, SWIVEL, WHEEL, or legacy ORDINARY")
endif()
set(CASTER_CONFIG_HEADER "${CASTER_WHEEL}.local.h")
if(CASTER_NODE_ID GREATER_EQUAL 4)
    set(CASTER_TYPE ordinary)
    set(CASTER_CONFIG_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/ordinary_caster/config")
    set(PROJECT_VER "0.5.0")
else()
    set(CASTER_TYPE ball)
    set(_ball_config_directory "${CMAKE_CURRENT_SOURCE_DIR}/ball_caster/config")
    set(_legacy_config_directory "${CMAKE_CURRENT_SOURCE_DIR}/config")
    if(EXISTS "${_ball_config_directory}/${CASTER_CONFIG_HEADER}" AND
       EXISTS "${_legacy_config_directory}/${CASTER_CONFIG_HEADER}")
        message(FATAL_ERROR "Ambiguous ball calibration: both ball_caster/config/${CASTER_CONFIG_HEADER} and config/${CASTER_CONFIG_HEADER} exist. Keep one authoritative private header; no file is selected automatically.")
    elseif(EXISTS "${_legacy_config_directory}/${CASTER_CONFIG_HEADER}")
        # Existing installations keep their original private calibration.
        set(CASTER_CONFIG_DIRECTORY "${_legacy_config_directory}")
    else()
        set(CASTER_CONFIG_DIRECTORY "${_ball_config_directory}")
    endif()
    set(PROJECT_VER "0.3.1")
endif()
if(NOT EXISTS "${CASTER_CONFIG_DIRECTORY}/${CASTER_CONFIG_HEADER}")
    message(FATAL_ERROR "Copy ${CASTER_CONFIG_DIRECTORY}/${CASTER_WHEEL}.example.h to ${CASTER_CONFIG_DIRECTORY}/${CASTER_CONFIG_HEADER} and configure it")
endif()
# The driver's signed 32-bit accumulator must wrap predictably; the application
# extends modular differences into int64_t without clearing the running counter.
add_compile_options(-fwrapv)
# Existing per-profile sdkconfig files retain selected Kconfig values over new
# defaults. Migrate only partition-table keys so every normal build/flash includes
# the new log partition. This edits build configuration, never device flash.
if(DEFINED SDKCONFIG)
    set(_caster_sdkconfig "${SDKCONFIG}")
else()
    set(_caster_sdkconfig "${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig")
endif()
if(EXISTS "${_caster_sdkconfig}")
    file(READ "${_caster_sdkconfig}" _caster_config_old)
    set(_caster_config_new "${_caster_config_old}")
    foreach(_choice SINGLE_APP SINGLE_APP_LARGE TWO_OTA TWO_OTA_LARGE)
        string(REPLACE "CONFIG_PARTITION_TABLE_${_choice}=y"
            "# CONFIG_PARTITION_TABLE_${_choice} is not set" _caster_config_new "${_caster_config_new}")
    endforeach()
    string(REPLACE "# CONFIG_PARTITION_TABLE_CUSTOM is not set"
        "CONFIG_PARTITION_TABLE_CUSTOM=y" _caster_config_new "${_caster_config_new}")
    string(REGEX REPLACE "CONFIG_PARTITION_TABLE_(CUSTOM_FILENAME|FILENAME)=\"[^\"]*\""
        "CONFIG_PARTITION_TABLE_\\1=\"partitions.csv\"" _caster_config_new "${_caster_config_new}")
    if(NOT _caster_config_new STREQUAL _caster_config_old)
        file(WRITE "${_caster_sdkconfig}" "${_caster_config_new}")
    endif()
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(hamr_caster_encoders)
if(NOT "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}" VERSION_EQUAL "5.4.2")
    message(FATAL_ERROR "Use the pinned ESP-IDF v5.4.2 SDK")
endif()
